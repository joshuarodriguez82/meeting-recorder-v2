# CPU-only dependency set for the app venv created by the first-launch

# REAL Python 3.13 install, not the embeddable distribution — the
# embeddable approach is abandoned (see HANDOFF.md bug #1,2).
#
# Version pins match the v1 stack that works on the dev PC
# (C:\meeting_recorder\.venv\). On a regular Python install, lightning
# 2.6.1 + torch 2.6.0 does NOT hit the speechbrain LazyModule recursion
# — that bug was specific to Python embeddable distributions.
#
#   torch==2.6.0 + torchaudio==2.6.0       (pyannote 3.3.2 compat)
#   speechbrain==1.0.3                     (1.1.0 has broken lazy imports)
#   pytorch-lightning==2.6.1 + lightning==2.6.1  (v1 stack; works on real Python)
#   numpy==2.1.3                           (torch 2.6 compat)

# Pyannote.audio 3.3.2 needs huggingface_hub <1 for the `use_auth_token=`
# keyword (removed in 1.0). The previous pin of ==0.23.0 was too strict:
# Python 3.13 has no wheel for tokenizers 0.19.x, which is what older
# transformers versions require (and pip backtracks to those when
# hf_hub is held at exactly 0.23.0 — pyo3 0.21.2 in the tokenizers sdist
# references PyUnicode_FromKindAndData / PyUnicode_4BYTE_KIND which were
# removed in 3.13, so the build fails). Allow the full 0.x line so pip
# can pick a newer hf_hub that ships py313 wheels for everything
# downstream.
huggingface_hub>=0.23,<1.0
torch==2.6.0
torchaudio==2.6.0
pyannote.audio==3.3.2
# pyannote.metrics 4.x requires numpy>=2.2.2, which contradicts the
# numpy==2.1.3 pin above (torch 2.6 compat) — and pyannote.audio 3.3.2
# leaves it unpinned, so a FRESH resolve floats into 4.x and the whole
# install becomes unsatisfiable. 3.2.1 is the last release compatible
# with this numpy. It pulls `docopt`, which is sdist-only, so the
# bootstrap exempts docopt from --only-binary alongside
# antlr4-python3-runtime (see lib.rs). Revisit when pyannote.audio
# ships a release that moves off the numpy 2.1 era.
pyannote.metrics<4

speechbrain==1.0.3
pytorch-lightning==2.6.1
lightning==2.6.1
pywin32
