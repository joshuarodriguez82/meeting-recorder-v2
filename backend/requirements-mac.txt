# macOS / Linux dependency set for the app venv created by the first-launch
# bootstrap (src-tauri/src/lib.rs → bootstrap_app_venv). Mirrors
# requirements-cpu.txt minus the Windows-only deps and plus pyobjc for
# EventKit (Calendar) access on macOS.
#
# Version pins match the v1 stack that works on the dev PC. The same
# versions install cleanly on Apple Silicon (universal2 wheels are
# available for torch 2.6 / numpy 2.1).
#
#   torch==2.6.0 + torchaudio==2.6.0       (pyannote 3.3.2 compat;
#                                           Apple Silicon uses MPS backend)
#   speechbrain==1.0.3                     (1.1.0 has broken lazy imports)
#   pytorch-lightning==2.6.1 + lightning==2.6.1
#   numpy==2.1.3                           (torch 2.6 compat)

uvicorn[standard]>=0.30.0

anthropic
openai
# Engagement-register Excel export (Phase 3); imported lazily.

# OS keychain for API keys. macOS Keychain on Mac, Secret Service on
# Linux. See backend/config/secrets.py.

sounddevice

scipy
faster-whisper
# Pyannote.audio 3.3.2 needs huggingface_hub <1 for the `use_auth_token=`
# keyword (removed in 1.0). The previous pin of ==0.23.0 was too strict:
# Python 3.13 has no wheel for tokenizers 0.19.x, which is what older
# transformers versions require (and pip backtracks to those when
# hf_hub is held at exactly 0.23.0 — pyo3 0.21.2 in the tokenizers sdist
# references PyUnicode_FromKindAndData / PyUnicode_4BYTE_KIND which were
# removed in 3.13, so the build fails). Allow the full 0.x line so pip
# can pick a newer hf_hub that ships py313 wheels for everything
# downstream.
huggingface_hub>=0.23,<1.0
torch==2.6.0
torchaudio==2.6.0
pyannote.audio==3.3.2
speechbrain==1.0.3
pytorch-lightning==2.6.1
lightning==2.6.1
numpy==2.1.3

# Semantic search across meetings — small (22 MB) MiniLM-L6 model,
# CPU-only, runs locally. Adds ~50ms per chunk on Apple Silicon.
sentence-transformers>=2.7.0,<3
# Drives the user's installed Chrome (channel='chrome') to scrape
# Outlook Web's day-view calendar for the Today tab's daily-brief
# sync. We do NOT bundle Chromium (~150 MB); see
# services/outlook_web_scraper.py for the channel='chrome' rationale.
# Persistent profile at <userdata>/web-session keeps the user signed
# in across launches; weekly MFA re-auth is expected.
playwright>=1.40
# pyannote.audio 3.3.2 uses huggingface_hub<1's `use_auth_token=` API.
# sentence-transformers 3+ pulls transformers 5+ which requires
# huggingface_hub>=1.5 and breaks pyannote at import. Pin both upper
# bounds until pyannote ships a fix.
transformers<5
